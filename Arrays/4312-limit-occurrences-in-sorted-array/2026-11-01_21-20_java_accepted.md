## Solution

```java
class Solution {
    public int[] limitOccurrences(int[] nums, int k) {
        ArrayList<Integer> list = new ArrayList<Integer>();
        int cnt = 1;
        
        list.add(nums[0]);
        for(int i = 1; i < nums.length; i++){
            if(nums[i] == nums[i - 1])
                cnt++;
            else
                cnt = 1;

            if(cnt <= k)
                list.add(nums[i]);
        }

        int n = list.size();
        int[] res = new int[n];
        for(int i = 0; i < n; i++){
            res[i] = list.get(i);
        }
        return res;
    }
}
```
